```lean
TVA.FloatV.AddRel.err {f : TVA.Format} (hf : f.Regular) (hs : f.specials ≠ TVA.Format.Specials.finite)
  (hu : f.u ≤ 1 / 16) {fusion : Bool} {x y r : TVA.FloatV f} (h : TVA.AddRel fusion x y r) (hx : x.isFinite = true)
```

reduction of products (`tl.sumc`).

```lean
  (h : ((a.mul b).add c).NaNEq r ∨ fusion = true ∧ (a.fma b c).NaNEq r) (ha : a.isFinite = true)
  (hb : b.isFinite = true) (hc : c.isFinite = true) (hM : |a.toRat * b.toRat| + |c.toRat| ≤ f.maxVal / 2 - 2 * f.eta) :
  r.isFinite = true ∧

```
